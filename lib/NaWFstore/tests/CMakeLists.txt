cmake_minimum_required(VERSION 3.16)
project(NaWFstoreTests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
add_executable(core_test core_test.cpp ../src/core/manager.cpp)
target_include_directories(core_test PRIVATE ../include)
add_test(NAME core COMMAND core_test)
add_executable(single_profile_test core_test.cpp ../src/core/manager.cpp)
target_include_directories(single_profile_test PRIVATE ../include)
target_compile_definitions(single_profile_test PRIVATE NAWFSTORE_MAX_PROFILES=1)
add_test(NAME single_profile COMMAND single_profile_test)
add_executable(nvs_test nvs_test.cpp ../src/core/manager.cpp ../ports/esp_idf/nvs_store.cpp)
target_include_directories(nvs_test PRIVATE fakes ../include)
add_test(NAME nvs COMMAND nvs_test)

target_include_directories(console_test PRIVATE ../include)
add_test(NAME console COMMAND console_test)
