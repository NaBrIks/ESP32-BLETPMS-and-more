option(NAWFSTORE_WITH_NVS "Persist profiles in ESP-IDF NVS" ON)
option(NAWFSTORE_WITH_CONSOLE "Build portable console adapter" ON)
option(NAWFSTORE_WITH_WEB "Enable application web adapter" ON)
option(NAWFSTORE_WITH_SOFTAP "Enable ESP-IDF SoftAP provisioning" ON)
option(NAWFSTORE_WITH_BOOTSTRAP "Enable application bootstrap profiles" ON)
set(NAWFSTORE_MAX_PROFILES 8 CACHE STRING "Maximum stored profiles (1..8)")
set(nawfstore_sources src/core/manager.cpp ports/esp_idf/wifi_driver.cpp)
set(nawfstore_requirements esp_wifi esp_event esp_netif esp_timer)
if(NAWFSTORE_WITH_NVS)
    list(APPEND nawfstore_sources ports/esp_idf/nvs_store.cpp)
    list(APPEND nawfstore_requirements nvs_flash)
endif()
if(NAWFSTORE_WITH_CONSOLE)
    list(APPEND nawfstore_sources adapters/console/console.cpp)
endif()
idf_component_register(SRCS ${nawfstore_sources} INCLUDE_DIRS include REQUIRES ${nawfstore_requirements})
target_compile_features(${COMPONENT_LIB} PUBLIC cxx_std_20)
foreach(feature NVS CONSOLE WEB SOFTAP BOOTSTRAP)
    target_compile_definitions(${COMPONENT_LIB} PUBLIC NAWFSTORE_WITH_${feature}=$<BOOL:${NAWFSTORE_WITH_${feature}}>)
endforeach()
target_compile_definitions(${COMPONENT_LIB} PUBLIC NAWFSTORE_MAX_PROFILES=${NAWFSTORE_MAX_PROFILES})
